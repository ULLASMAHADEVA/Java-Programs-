import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int A = 0, B = 0, C = 0, D = 0, F = 0, total = 0;

        for (int i = 0; i < n; i++) {
            int m = sc.nextInt();
            total += m;

            switch (m / 10) {
                case 10:
                case 9:
                    System.out.print("A ");
                    A++;
                    break;
                case 8:
                    System.out.print("B ");
                    B++;
                    break;
                case 7:
                    System.out.print("C ");
                    C++;
                    break;
                case 6:
                    System.out.print("D ");
                    D++;
                    break;
                default:
                    System.out.print("F ");
                    F++;
            }
        }

        System.out.println();
        System.out.println("A=" + A + ", B=" + B + ", C=" + C + ", D=" + D + ", F=" + F);
        System.out.println("Average = " + (double) total / n);
    }
}
