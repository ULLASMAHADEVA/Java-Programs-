import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n];

        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();

        int largest = a[0];
        int second = 0;
        boolean found = false;

        for (int i = 1; i < n; i++) {
            if (a[i] > largest) {
                second = largest;
                largest = a[i];
                found = true;
            } 
            else if (a[i] < largest && (!found || a[i] > second)) {
                second = a[i];
                found = true;
            }
        }

        if (found)
            System.out.println("Second Largest = " + second);
        else
            System.out.println("Second largest element does not exist");
    }
}
